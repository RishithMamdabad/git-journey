import torch

from transformers import pipeline
from sentence_transformers import SentenceTransformer

from config import (
    SENTIMENT_MODEL,
    EMBEDDING_MODEL
)

# =====================================================
# PERFORMANCE
# =====================================================

torch.set_num_threads(4)

DEVICE = (

    0

    if torch.cuda.is_available()

    else -1
)

# =====================================================
# LOAD MODELS
# =====================================================

@st.cache_resource
def get_models():

    sentiment_model = pipeline(

        task="sentiment-analysis",

        model=SENTIMENT_MODEL,

        device=DEVICE,

        truncation=True
    )

    embedding_model = SentenceTransformer(

        EMBEDDING_MODEL,

        device=(

            "cuda"

            if torch.cuda.is_available()

            else "cpu"
        )
    )

    embedding_model.max_seq_length = 256

    return (

        sentiment_model,
        embedding_model
    )

    return {

        "device": (

            torch.cuda.get_device_name(0)

            if torch.cuda.is_available()

            else "CPU"
        )
    }

import re

# =====================================================
# CLEAN TEXT
# =====================================================

def clean_text(text):

    text = str(text).lower()

    text = re.sub(
        r"http\S+|www\S+",
        "",
        text
    )

    text = re.sub(
        r"<.*?>",
        "",
        text
    )

    text = re.sub(
        r"[^a-zA-Z0-9\s]",
        " ",
        text
    )

    text = re.sub(
        r"\s+",
        " ",
        text
    ).strip()

    return text

# =====================================================
# DOMAIN DETECTION
# =====================================================

def detect_domain(df):

    columns = " ".join(
        df.columns.astype(str)
    ).lower()

    best_domain = "general"
    best_score = 0

    for domain, keywords in DOMAIN_KEYWORDS.items():

        score = sum(
            1 for keyword in keywords
            if keyword in columns
        )

        if score > best_score:

            best_score = score
            best_domain = domain

    return best_domain

# =====================================================
# RATING COLUMN DETECTION
# =====================================================

def is_rating_column(series):

    try:

        values = (
            pd.to_numeric(
                series.dropna(),
                errors="coerce"
            ).dropna()
        )

        if values.empty:
            return False

        return values.between(1, 5).all()

    except:
        return False

# =====================================================

# =====================================================

        "email",
        "location",
        "salary",
        "code",
        "name"
    ]

    if any(
        keyword in column_name
        for keyword in skip_keywords
    ):
        return False

    texts = values.tolist()

    meaningful_count = 0

    for text in texts:

        clean = clean_text(text)

        words = clean.split()

        if len(words) >= MIN_WORDS_FOR_NLP:
            meaningful_count += 1

    ratio = meaningful_count / len(texts)

    return ratio >= MIN_MEANINGFUL_RATIO

# =====================================================
# RATING → SENTIMENT
# =====================================================

def rating_to_sentiment(value):

    try:

        value = round(float(value))

        value = min(max(value, 1), 5)

        return RATING_SENTIMENT.get(
            value,
            "NEUTRAL"
        )

    except:
        return "NEUTRAL"

# =====================================================
# MAP SENTIMENT
# =====================================================

def map_sentiment(label, score):

    if score < SENTIMENT_THRESHOLD:
        return "NEUTRAL"

    if label.upper() == "POSITIVE":
        return "POSITIVE"

    if label.upper() == "NEGATIVE":
        return "NEGATIVE"

    return "NEUTRAL"

# =====================================================
# SENTIMENT ANALYSIS
# =====================================================

def get_sentiment_batch(texts):

    cleaned_texts = [
        clean_text(t)
        for t in texts
    ]

    results = sentiment_model(
        cleaned_texts,
        truncation=True,
        batch_size=32
    )

    sentiments = []
    scores = []

    for result in results:

        sentiment = map_sentiment(
            result["label"],
            result["score"]
        )

        if sentiment == "NEUTRAL":
            score = 0
        else:
            score = round(
                result["score"] * 100,
                2
            )

        sentiments.append(sentiment)
        scores.append(score)

    return sentiments, scores

# =====================================================
# TOPIC MODEL
# =====================================================

def get_topic_batch(texts):

    topics = ENTERPRISE_TOPICS

    topic_embeddings = embedding_model.encode(
        topics,
        convert_to_tensor=True
    )

    cleaned_texts = [
        clean_text(t)
        for t in texts
    ]

    text_embeddings = embedding_model.encode(
        cleaned_texts,
        convert_to_tensor=True
    )

    assigned_topics = []

    for emb in text_embeddings:

        scores = util.cos_sim(
            emb,
            topic_embeddings
        )[0]

        best_idx = int(scores.argmax())

        topic = topics[best_idx]

        assigned_topics.append(topic)

    return assigned_topics

# =====================================================
# PROCESS DATAFRAME
# =====================================================

def process_dataframe(df):

    all_topics = []

    for col in df.columns:

        if not should_analyze_column(df[col]):
            continue

        texts = (
            df[col]
            .fillna("")
            .astype(str)
            .tolist()
        )

        if is_rating_column(df[col]):

            sentiment_vals = [
                rating_to_sentiment(v)
                for v in texts
            ]

            df[f"{col}_sentiment"] = sentiment_vals

            continue

        valid_mask = []

        for t in texts:

            clean = clean_text(t)

            words = clean.split()

            valid_mask.append(
                len(words) >= MIN_WORDS_FOR_NLP
            )

        valid_texts = [
            t
            for t, valid in zip(texts, valid_mask)
            if valid
        ]

        topic_vals = ["General"] * len(texts)

        sentiment_vals = ["NEUTRAL"] * len(texts)

        sentiment_scores = [0] * len(texts)

        if valid_texts:

            topics = get_topic_batch(valid_texts)

            sentiments, scores = (
                get_sentiment_batch(valid_texts)
            )

            idx = 0

            for i, valid in enumerate(valid_mask):

                if valid:

                    topic_vals[i] = topics[idx]

                    sentiment_vals[i] = sentiments[idx]

                    sentiment_scores[i] = scores[idx]

                    idx += 1

        df[f"{col}_topic"] = topic_vals

        df[f"{col}_sentiment"] = sentiment_vals

        df[f"{col}_sentiment_score"] = sentiment_scores

        all_topics.extend(topic_vals)

    sentiment_cols = [

        c for c in df.columns
        if (
            c.endswith("_sentiment")
            and c != "overall_sentiment"
        )
    ]

    overall_sentiments = []

    for _, row in df.iterrows():

        sentiments = [
            row[c]
            for c in sentiment_cols
        ]

        positive = sentiments.count("POSITIVE")

        negative = sentiments.count("NEGATIVE")

        if positive > negative:

            overall = "POSITIVE"

        elif negative > positive:

            overall = "NEGATIVE"

        else:

            overall = "NEUTRAL"

        overall_sentiments.append(overall)

    df["overall_sentiment"] = overall_sentiments

    return df, all_topics

def generate_file_insights(df):

    sentiments = (
        df["overall_sentiment"]
        .dropna()
        .tolist()
    )

    positive = sentiments.count("POSITIVE")

    negative = sentiments.count("NEGATIVE")

    neutral = sentiments.count("NEUTRAL")

    overall = (
        "POSITIVE"
        if positive >= negative
        else "NEGATIVE"
    )

    return {

        "total_records": len(df),

        "overall_sentiment": overall,

        "positive": positive,

        "negative": negative,

        "neutral": neutral
    }

        if c.endswith("_topic")

    for t_col in topic_cols:

        base = t_col.replace(
            "_topic",
            ""
        )

        s_col = f"{base}_sentiment"

        if s_col not in df.columns:
            continue

        temp = pd.DataFrame({

            "Topic": df[t_col],

            "Sentiment": df[s_col]
        })

        negative_df = temp[
            temp["Sentiment"] == "NEGATIVE"
        ]

        negative_topics.extend(
            negative_df["Topic"].tolist()
        )

    if not negative_topics:

        return pd.DataFrame(
            columns=[
                "Topic",
                "Frequency"
            ]
        )

    negative_df = (
        pd.Series(negative_topics)
        .value_counts()
        .head(5)
        .reset_index()
    )

    negative_df.columns = [
        "Topic",
        "Frequency"
    ]

    return negative_df

def generate_risk_analysis(
    processed_df,
    dept_df=None
):

    risks = []

    negative_topics = []

    topic_cols = [

        c for c in processed_df.columns
        if c.endswith("_topic")
    ]

    for t_col in topic_cols:

        base = t_col.replace(
            "_topic",
            ""
        )

        s_col = f"{base}_sentiment"

        if s_col not in processed_df.columns:
            continue

        temp = processed_df[
            processed_df[s_col] == "NEGATIVE"
        ]

        negative_topics.extend(

            temp[t_col]
            .dropna()
            .astype(str)
            .tolist()
        )

    top_negative = Counter(
        negative_topics
    ).most_common(3)

    if top_negative:

        risk_topics = ", ".join([
            t[0]
            for t in top_negative
        ])

        risks.append(

            f"The strongest negative sentiment signals across the dataset are centered around {risk_topics}. These themes indicate operational dissatisfaction, experience gaps, and declining engagement patterns that require immediate leadership intervention through process optimization, stronger communication systems, and accountability frameworks."
        )

    if dept_df is not None and not dept_df.empty:

        highest_risk_department = None
        highest_negative = -1

        for _, row in dept_df.iterrows():

            negative_count = row.get(
                'NEGATIVE',
                0
            )

            if negative_count > highest_negative:

                highest_negative = negative_count

                highest_risk_department = row.iloc[0]

        if highest_risk_department:

            risks.append(

                f"The '{highest_risk_department}' department currently demonstrates the highest concentration of negative sentiment. Leadership should prioritize operational reviews, workforce engagement initiatives, management effectiveness evaluations, and cross-functional collaboration improvements within this department."
            )

    designation_df = generate_designation_sentiment(
        processed_df
    )

    if not designation_df.empty:

        highest_risk_position = None
        highest_negative = -1

        for _, row in designation_df.iterrows():

            negative_count = row.get(
                'NEGATIVE',
                0
            )

            if negative_count > highest_negative:

                highest_negative = negative_count

                highest_risk_position = row.iloc[0]

        if highest_risk_position:

            risks.append(

                f"The '{highest_risk_position}' position group is reflecting elevated negative sentiment levels. This suggests workload pressure, leadership alignment gaps, career progression concerns, or operational inefficiencies affecting this workforce segment."
            )

    return risks[:3]
